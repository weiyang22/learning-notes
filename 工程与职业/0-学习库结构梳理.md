## 11. 附录 B: 统计脚本 (可复用)

### 11.1 一键扫描脚本

```python
#!/usr/bin/env python3
"""scan_md.py - 学习库 MD 统计 + 异常检测

用法: python scan_md.py
"""
import os
import sys
from collections import defaultdict, Counter
from pathlib import Path

ROOT = "/Users/weiyang/Desktop/个人资料/资料/学习"

def scan():
    counts = defaultdict(int)
    sizes = defaultdict(int)
    naming = Counter()
    frontmatter_total = 0
    frontmatter_files = []
    no_frontmatter_files = []
    bak_files = []
    plan_files = []

    for root, dirs, files in os.walk(ROOT):
        for f in files:
            if not f.endswith(".md"):
                continue
            path = Path(root) / f
            rel = path.relative_to(ROOT)

            # 一级目录
            top = rel.parts[0] if len(rel.parts) > 1 else "根目录"
            counts[top] += 1
            sizes[top] += path.stat().st_size

            # 命名规范
            if f.startswith("第") and "章" in f:
                if "-NN-" in f or len(f.split("-")) > 2:
                    naming["第xx章-NN-标题"] += 1
                else:
                    naming["第xx章-标题"] += 1
            elif f.startswith("附录"):
                naming["附录X-标题"] += 1
            elif f.startswith("0-"):
                naming["索引/总目录"] += 1
            elif f.endswith(".bak"):
                bak_files.append(str(rel))
            else:
                naming["主题/其他"] += 1

            # frontmatter 检查
            with open(path, encoding="utf-8") as fh:
                first_line = fh.readline().strip()
            if first_line == "---":
                frontmatter_total += 1
                frontmatter_files.append(str(rel))
            else:
                no_frontmatter_files.append(str(rel))

            # plan.md
            if "plan" in f.lower():
                plan_files.append(str(rel))

    return counts, sizes, naming, frontmatter_total, no_frontmatter_files, bak_files, plan_files


def main():
    counts, sizes, naming, fm, no_fm, baks, plans = scan()
    total = sum(counts.values())
    total_size = sum(sizes.values())

    print(f"=== 学习库统计 ===")
    print(f"总 MD: {total}")
    print(f"总大小: {total_size/1e6:.1f} MB")
    print(f"含 frontmatter: {fm} ({fm/total*100:.1f}%)")
    print(f"无 frontmatter: {len(no_fm)} ({len(no_fm)/total*100:.1f}%)")
    print()

    print(f"=== 一级目录 ===")
    for k, v in sorted(counts.items(), key=lambda x: -x[1]):
        pct = v / total * 100
        print(f"  {k:30s} {v:>5} md  {sizes[k]/1e6:>6.1f} MB  {pct:>5.1f}%")

    print(f"\n=== 命名规范 ===")
    for k, v in naming.most_common():
        print(f"  {k:30s} {v:>5}")

    if baks:
        print(f"\n=== .bak 残留 ===")
        for b in baks:
            print(f"  {b}")

    if plans:
        print(f"\n=== plan* 文件 (避免冲突) ===")
        for p in plans:
            print(f"  {p}")


if __name__ == "__main__":
    main()
```

### 18.5 frontmatter 自动校验脚本

```python
#!/usr/bin/env python3
"""check_frontmatter.py - 校验所有 md 的 frontmatter 完整性"""
import os
import re
import yaml
from pathlib import Path

ROOT = "/Users/weiyang/Desktop/个人资料/资料/学习"

def check_frontmatter():
    required_keys = ["title", "last_updated", "status", "tags", "description"]
    issues = []

    for root, _, files in os.walk(ROOT):
        for f in files:
            if not f.endswith(".md"):
                continue
            path = Path(root) / f

            with open(path, encoding="utf-8") as fh:
                content = fh.read()

            # 检查 frontmatter 存在
            if not content.startswith("---\n"):
                issues.append(("MISSING", str(path)))
                continue

            # 解析
            try:
                fm_end = content.index("\n---\n", 4)
                fm_text = content[4:fm_end]
                fm = yaml.safe_load(fm_text)
            except (ValueError, yaml.YAMLError) as e:
                issues.append(("INVALID", str(path), str(e)))
                continue

            # 检查必填字段
            for key in required_keys:
                if key not in fm:
                    issues.append(("MISSING_KEY", str(path), key))

    return issues
```

    """计算文档库的"健康指数" 0-100"""
    total_md = 0

    bak = 0

            total_md += 1
            path = Path(r) / f

                content = fh.read(1000)
            if content.startswith("---\n"):
                with_fm += 1

            if f.startswith("_"):
                bad_naming += 1

            # 检查重复

                dup_files += 1
            seen_names[f] = path

    score = 100
