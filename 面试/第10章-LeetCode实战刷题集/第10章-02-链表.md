```python
def reverseList(head):
    prev, curr = None, head
    while curr:
        nxt = curr.next
        curr.next = prev
        prev = curr
        curr = nxt
    return prev
```

### 1.3 LeetCode 25 - K 个一组翻转 (Hard)

```python
def reverseKGroup(head, k):
    dummy = ListNode(0)
    dummy.next = head
    pre = dummy
    while True:
        # 检查剩余是否够 k 个
        kth = pre
        for _ in range(k):
            kth = kth.next
            if not kth: return dummy.next
        # 翻转 k 个
        start = pre.next
        end = kth
        nxt = end.next
        # 翻转

        while curr is not nxt:

            curr.next = prev
            prev = curr
            curr = tmp
        # 接回
        pre.next = prev
        start.next = nxt
        pre = start
```

### 4.4 LeetCode 61 - 旋转链表 (Medium)

```python
def rotateRight(head, k):
    if not head or k == 0: return head
    # 计算长度 + 形成环
    cur, n = head, 1
    while cur.next:
        cur = cur.next
        n += 1
    cur.next = head  # 成环
    k = k % n
    # 找新尾
    for _ in range(n - k):
        cur = cur.next
    new_head = cur.next
    cur.next = None
    return new_head
```

## A.1 基础结构

```python
class ListNode:
    """单向链表节点"""
    __slots__ = ('val', 'next')

    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

    def __repr__(self):
        nxt = self.next.val if self.next else 'None'
        return f'ListNode({self.val}) -> {nxt}'


class DoubleNode:
    """双向链表节点 (用于 LRU / LFU 设计题)"""
    __slots__ = ('key', 'val', 'prev', 'next')

    def __init__(self, key=0, val=0, prev=None, next=None):
        self.key = key
        self.val = val
        self.prev = prev
        self.next = next


class RandomNode:
    """带随机指针的链表节点 (LeetCode 138)"""
    __slots__ = ('val', 'next', 'random')

    def __init__(self, val=0, next=None, random=None):
        self.val = val
        self.next = next
        self.random = random
```

```python
def build_list(arr):

    if not arr:

    head = ListNode(arr[0])
    cur = head

def to_list(head):
    """链表转数组 (调试用)"""
    res = []
    cur = head

def print_list(head, max_n=20):
    """可视化打印 (避免无限循环)"""
    cur = head
    seen = set()
    res = []
    n = 0
    while cur and n < max_n:
        if id(cur) in seen:
            res.append(f'...cycle @ {cur.val}')
            break
        seen.add(id(cur))
        res.append(str(cur.val))
        cur = cur.next
        n += 1
    print(' -> '.join(res))


def make_cycle(head, pos):
    """把链表尾部接到第 pos 个节点 (-1 表示不接, 无环)"""
    if pos < 0:
        return head
    cycle_entry = None
    cur, idx = head, 0
    tail = None
    while cur:
        if idx == pos:
            cycle_entry = cur
        tail = cur
        cur = cur.next
        idx += 1
    if tail and cycle_entry:
        tail.next = cycle_entry
    return head
```

## A.3 六大基本模式

### 模式 1: 哑节点 (Dummy Node)

```python
def with_dummy_template(head, transformer):
    """所有"修改头节点"操作的标准套路"""
    dummy = ListNode(0, head)
    pre = dummy
    cur = head
    while cur:
        # 业务逻辑: 跳过/删除/插入
        if SHOULD_DELETE:
            pre.next = cur.next
            cur = cur.next
        elif SHOULD_INSERT:
            new_node = ListNode(v)
            new_node.next = cur.next
            cur.next = new_node
            pre = new_node  # 取决于需求
            cur = cur.next
        else:
            pre = cur
            cur = cur.next
    return dummy.next
```

**哑节点使用准则**:
- 可能需要删除头节点时 → 必须用 dummy
- 反转/合并时统一入口 → 推荐用 dummy
- 单纯遍历查找 → 不需要 dummy

**典型应用**: 19, 21, 23, 24, 82, 203, 707, 1171

### 模式 2: 快慢指针 (Fast & Slow)

```python
def fast_slow_template(head):
    """快慢指针六大变种"""
    slow, fast = head, head
    variants = {
        'midpoint': (fast and fast.next, 1),       # 876
        'entry_of_cycle': None,                    # 142
        'kth_from_end': (fast.next is None, k),    # 19
        'palindrome_check': (fast.next and fast.next.next, None),  # 234
        'cycle_detect': (fast and fast.next, None),# 141
        'intersect': None,                         # 160 (双端双指针)
    }
    return slow, fast
```

**六类典型场景**:

| 场景 | 描述 | 实现要点 | 例题 |
|------|------|----------|------|
| 中点 | slow 走一步, fast 走两步 | 退出时 slow 即为中点 | 876 |
| 入口检测 | 相遇后 head 再出发 | slow 与新指针相遇点即入口 | 142 |
| 倒数第 K | fast 先走 K 步 | 同时出发, fast 到尾时 slow 即所求 | 19 |
| 回文 | 中点 + 反转 + 比较 | 比较后需恢复链表(或注明) | 234 |
| 环检测 | slow 与 fast 相遇即可 | 注意 next 域检查 | 141 |
| 相交 | 双指针 tail-redirect | 都遍历 headA+headB 后必相遇 | 160 |

**数学证明 (环入口)**:

设链表头到环入口距离 `a`, 环长 `b`, slow 相遇时走过 `s`, fast 走过 `f=2s`, 由 `f = s + nb`:
```text
2s - s = nb → s = nb
head 走到入口: a 步, slow 此时相遇点再走 (a - nb - 0+a)...
简化为: head 走 a 步到入口, slow 从相遇走 (a - nb + a) → ... → a
```
具体: 此时让 head 和 slow 同速出发, 都走 a 步同时到达入口, 故相遇点即入口。

### 模式 4: 合并 (Merge)

```python
# 两路归并 (有序)
def merge_two(a, b):
    dummy = ListNode(0)
    cur = dummy
    while a and b:
        if a.val <= b.val:
            cur.next = a
            a = a.next
        else:
            cur.next = b
            b = b.next
        cur = cur.next
    cur.next = a or b
    return dummy.next

# K 路归并 (优先队列)
import heapq
def merge_k(lists):
    dummy = ListNode(0)
    cur = dummy
    heap = [(l.val, i, l) for i, l in enumerate(lists) if l]
    heapq.heapify(heap)
    while heap:
        v, i, node = heapq.heappop(heap)
        cur.next = node
        cur = cur.next
        if node.next:
            heapq.heappush(heap, (node.next.val, i, node.next))
    return dummy.next

# 自底向上归并 (O(1) 空间, 排序链表)
def merge_sort(head):
    if not head or not head.next:
        return head
    # 1. 切两半
    slow, fast = head, head.next
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
    mid = slow.next
    slow.next = None
    # 2. 递归
    left = merge_sort(head)
    right = merge_sort(mid)
    # 3. 合并
    return merge_two(left, right)
```

```python
def three_sum_linked(head, target):
    nums = to_list(head)
    nums.sort()
    res = []
    n = len(nums)
    for i in range(n - 2):
        if i > 0 and nums[i] == nums[i-1]:
            continue
        lo, hi = i + 1, n - 1
        while lo < hi:
            s = nums[i] + nums[lo] + nums[hi]
            if s == target:
                res.append([nums[i], nums[lo], nums[hi]])
                while lo < hi and nums[lo] == nums[lo+1]: lo += 1
                while lo < hi and nums[hi] == nums[hi-1]: hi -= 1
                lo += 1; hi -= 1
            elif s < target:
                lo += 1
            else:
                hi -= 1
    return res
```

## B.4 环检测: Floyd vs 哈希

| 方法 | 时间 | 空间 | 实现 | 适用 |
|------|------|------|------|------|
| 快慢指针 | O(n) | O(1) | 略 | **首选** |

```python
def hasCycle_hash(head):
    visited = set()
    cur = head
    while cur:
        if id(cur) in visited:
            return True
        visited.add(id(cur))
        cur = cur.next
```

## B.6 LRU 缓存 (LeetCode 146): 哈希+双向链表 vs OrderedDict

```python
# 解法 1: 手写双向链表 (面试推荐, 展示底层)
class LRUCache:
    def __init__(self, capacity):
        self.cap = capacity
        self.cache = {}
        self.head = self.tail = DoubleNode()
        self.head.next = self.tail
        self.tail.prev = self.head

    def _remove(self, node):
        node.prev.next = node.next
        node.next.prev = node.prev

    def _add_to_head(self, node):
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node

    def get(self, key):
        if key not in self.cache:
            return -1
        node = self.cache[key]
        self._remove(node)
        self._add_to_head(node)
        return node.val

    def put(self, key, value):
        if key in self.cache:
            node = self.cache[key]
            node.val = value
            self._remove(node)
            self._add_to_head(node)
            return
        if len(self.cache) == self.cap:
            lru = self.tail.prev
            self._remove(lru)
            del self.cache[lru.key]
        node = DoubleNode(key, value)
        self.cache[key] = node
        self._add_to_head(node)

# 解法 2: collections.OrderedDict (工程首选, 简洁)
from collections import OrderedDict
class LRUCache_simple:
    def __init__(self, capacity):
        self.cap = capacity
        self.cache = OrderedDict()

    def get(self, key):
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)
        return self.cache[key]

    def put(self, key, value):
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        if len(self.cache) > self.cap:
            self.cache.popitem(last=False)
```

```python
            fast = fast.next.next

        if start >= end:
            return None

        left = build_size(start, mid)
        nonlocal cur
        root = TreeNode(cur.val)
        cur = cur.next
        root.left = left
        root.right = build_size(mid + 1, end)

        p = p.next
```

## Q1: 链表 vs 数组, 各自的优劣?

## Q2: 什么是哑节点 (Dummy Node), 何时用?

哑节点是 head 前面的占位节点 (值任意, 通常 0)。它:
1. **统一边界处理**: 删除头节点时无需特判
2. **统一合并/反转**: 返回 dummy.next 即新头
3. **减少 if-else 嵌套**

**适用**: 删除/合并/反转/区间操作
**不适用**: 仅查找/计数时, 用哑节点反而啰嗦

## Q3: 快慢指针的数学原理?

**找中点**: slow 走 1, fast 走 2 → fast 到尾时 slow 正好在中点
$$d_{slow} = d_{fast}/2 = n/2$$

**环检测**: 第一次相遇 `slow = nb`
**环入口**: 设 head 到入口距离 `a`
$$f = 2s \Rightarrow f - s = s = nb$$
再走 `a` 步: head 走了 a, slow 走了 nb + a, 模 b 同位 → 相遇于入口。

## Q4: 如何判断链表是否有环?

- **快慢指针 (Floyd)**: 必问, O(1) 空间
- **哈希 visited**: 工程备用, O(n) 空间
- **修改 val 为不可能值**: 破坏原链表, 慎用

## Q5: 反转链表的关键点?

1. **三指针轮转**: pre, cur, nxt
2. **cur.next 必须先保存**, 否则断链
3. **返回 pre** (新头), 不是 cur

## Q6: 为什么反转区间 [left, right] 用头插法?

把区间内节点逐个插入到 pre 之后, 局部反转:
- 时间 O(n)
- 比"全部反转再接回"少写代码, 且不需还原

## Q7: 合并 K 个链表的最小堆 vs 分治?

- **最小堆**: O(N log k), 维护当前 K 个候选
- **分治**: O(N log k), 两两配对合并, 借助合并 2 个链表的子过程

后者常数小 (堆操作较重), 实际更快。

## Q8: 链表归并排序的时间复杂度?

- **时间**: O(n log n) - 每层合并 O(n), 共 log n 层

## Q21: 旋转链表 (61) 的优雅做法?

把链表成环, 然后在新位置断开:

```python
def rotateRight(head, k):
    if not head or k == 0: return head
    # 成环
    cur, n = head, 1
    while cur.next: cur, n = cur.next, n+1
    cur.next = head
    # 走 n - k%n 步到新尾
    k = k % n
    for _ in range(n - k): cur = cur.next
    new_head = cur.next
    cur.next = None
    return new_head
```

## Q22: 设计一个能从头部/尾部增删的双端队列 (链表版)?

```python
class Deque:
    def __init__(self):
        self.front = self.rear = DoubleNode()
        self.size = 0
    def push_front(self, val):
        node = DoubleNode(val)
        node.next = self.front.next
        self.front.next.prev = node
        self.front.next = node
        node.prev = self.front
        self.size += 1
    def push_back(self, val):
        node = DoubleNode(val)
        node.prev = self.rear.prev
        self.rear.prev.next = node
        self.rear.prev = node
        node.next = self.rear
        self.size += 1
    def pop_front(self):
        if self.size == 0: return None
        node = self.front.next
        node.next.prev = self.front
        self.front.next = node.next
        self.size -= 1
        return node.val
    def pop_back(self):

        node.prev.next = self.rear
        self.rear.prev = node.prev
        self.size -= 1
```

## Q26: 链表如何实现 hashmap 的 chaining?

```python
class HashMap:
    def __init__(self, cap=1000):
        self.buckets = [None] * cap

    def _hash(self, key):
        return hash(key) % len(self.buckets)

    def put(self, key, val):
        idx = self._hash(key)
        head = self.buckets[idx]
        cur = head
        while cur:
            if cur.key == key:
                cur.val = val
                return
            cur = cur.next
        new = ListNode((key, val))
        new.next = head
        self.buckets[idx] = new

    def get(self, key):
        idx = self._hash(key)
        cur = self.buckets[idx]
        while cur:
            if cur.val[0] == key:
                return cur.val[1]
            cur = cur.next
        return None
```

## G.1 反转模板

```python
# 整条反转
def reverse(head):
    pre, cur = None, head
    while cur:
        nxt = cur.next
        cur.next = pre
        pre = cur
        cur = nxt
    return pre

# K 组反转
def reverseKGroup(head, k):
    dummy = ListNode(0, head)
    pre = dummy
    while True:
        kth = pre
        for _ in range(k):
            kth = kth.next
            if not kth: return dummy.next
        start = pre.next
        end = kth
        nxt = end.next
        # 反转 [start, end]
        prev, cur = nxt, start
        while cur is not nxt:
            tmp = cur.next
            cur.next = prev
            prev = cur
            cur = tmp
        pre.next = prev
        start.next = nxt
        pre = start
```

## G.2 合并模板

```python
def mergeKLists(lists):
    import heapq
    heap = [(l.val, i, l) for i, l in enumerate(lists) if l]
    heapq.heapify(heap)
    dummy = ListNode(0)
    cur = dummy
    while heap:
        v, i, node = heapq.heappop(heap)
        cur.next = node
        cur = cur.next
        if node.next:
            heapq.heappush(heap, (node.next.val, i, node.next))
    return dummy.next
```

## G.5 LRU 模板

```python
class LRUCache:
    def __init__(self, cap):
        self.cap = cap
        self.c = {}
        self.head = self.tail = DoubleNode()
        self.head.next = self.tail
        self.tail.prev = self.head

    def _rem(self, n):
        n.prev.next = n.next
        n.next.prev = n.prev

    def _add(self, n):
        n.next = self.head.next
        n.prev = self.head
        self.head.next.prev = n
        self.head.next = n

    def get(self, k):
        if k not in self.c: return -1
        n = self.c[k]
        self._rem(n); self._add(n)
        return n.val

    def put(self, k, v):
        if k in self.c:
            n = self.c[k]; n.val = v
            self._rem(n); self._add(n)
        else:
            if len(self.c) == self.cap:
                lru = self.tail.prev
                self._rem(lru); del self.c[lru.key]
            n = DoubleNode(k, v)
            self.c[k] = n; self._add(n)
```

## I.3 跳表 (Skip List)

链表 + 多级索引, 平均 O(log n) 查找, Redis zset 内部使用。

```python
class SkipList:
    class Node:
        def __init__(self, val):
            self.val = val
            self.next = [None] * 16  # 16 层
    def __init__(self):
        self.head = self.Node(-float('inf'))
    def find(self, target):
        cur = self.head
        for level in reversed(range(16)):
            while cur.next[level] and cur.next[level].val < target:
                cur = cur.next[level]
        return cur.next[0]
```
