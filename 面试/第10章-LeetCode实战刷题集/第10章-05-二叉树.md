```python
def inorderTraversal(root):
    res = []
    def dfs(node):
        if not node: return
        dfs(node.left)
        res.append(node.val)
        dfs(node.right)
    dfs(root)
```

### 1.2 LeetCode 144 - 二叉树的前序遍历 (Easy)

```python
def preorderTraversal(root):
    res = []
    stack = [root]
    while stack:
        node = stack.pop()
        if node:
            res.append(node.val)
            stack.append(node.right)
            stack.append(node.left)
    return res
```

### 1.3 LeetCode 145 - 二叉树的后序遍历 (Easy)

```python
def postorderTraversal(root):
    res = []
    stack = [root]
    while stack:
        node = stack.pop()
        if node:
            res.append(node.val)
            stack.append(node.left)
            stack.append(node.right)
    return res[::-1]
```

### 1.4 LeetCode 102 - 二叉树的层序遍历 (Medium)

```python
def levelOrder(root):
    from collections import deque
    if not root: return []
    res = []
    queue = deque([root])
    while queue:
        level = []
        for _ in range(len(queue)):
            node = queue.popleft()
            level.append(node.val)
            if node.left: queue.append(node.left)
            if node.right: queue.append(node.right)
        res.append(level)
    return res
```

```python
def pathSum(root, target):
    res = []
    def dfs(node, path, remain):
        if not node: return
        path.append(node.val)
        if not node.left and not node.right and remain == node.val:
            res.append(path[:])
        dfs(node.left, path, remain - node.val)
        dfs(node.right, path, remain - node.val)
        path.pop()
    dfs(root, [], target)
    return res
```

```python
        if not node: return 0
        l = max(0, dfs(node.left))
        r = max(0, dfs(node.right))
```

### 3.4 LeetCode 437 - 路径总和 III (Medium)

```python
def pathSum_iii(root, target):
    from collections import defaultdict
    cnt = defaultdict(int)
    cnt[0] = 1
    res = [0]
    def dfs(node, prefix):
        if not node: return
        prefix += node.val
        res[0] += cnt[prefix - target]
        cnt[prefix] += 1
        dfs(node.left, prefix)
        dfs(node.right, prefix)
        cnt[prefix] -= 1
    dfs(root, 0)
    return res[0]
```

## A.1 基础结构

```python
class TreeNode:
    """二叉树节点"""
    __slots__ = ('val', 'left', 'right')

    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

    def __repr__(self):
        return f'TreeNode({self.val})'


class NaryTreeNode:
    """N 叉树节点"""
    __slots__ = ('val', 'children')

    def __init__(self, val=0, children=None):
        self.val = val
        self.children = children if children is not None else []


class Codec:
    """二叉树序列化/反序列化 (297)"""
    def serialize(self, root):
        def dfs(node):
            if not node: return 'None'
            return f'{node.val},{dfs(node.left)},{dfs(node.right)}'
        return dfs(root)

    def deserialize(self, data):
        vals = iter(data.split(','))
        def dfs():
            v = next(vals)
            if v == 'None': return None
            node = TreeNode(int(v))
            node.left = dfs()
            node.right = dfs()
            return node
        return dfs()
```

```python
def build_tree(arr):
    """从层序数组构造二叉树 (-1 表示 None)"""
    if not arr or arr[0] == -1:
        return None
    root = TreeNode(arr[0])
    queue = [root]
    i = 1
    while queue and i < len(arr):
        node = queue.pop(0)
        if i < len(arr) and arr[i] != -1:
            node.left = TreeNode(arr[i])
            queue.append(node.left)
        i += 1
        if i < len(arr) and arr[i] != -1:
            node.right = TreeNode(arr[i])
            queue.append(node.right)
        i += 1
    return root


def tree_to_array(root):
    """二叉树转层序数组 (-1 表示空位, 直到末尾 None)"""
    if not root:
        return []
    result = []
    queue = [root]
    not_all_none = True
    while queue and not_all_none:
        level = []
        not_all_none = False
        for _ in range(len(queue)):
            node = queue.pop(0)
            if node:
                level.append(node.val)
                queue.append(node.left)
                queue.append(node.right)
                if node.left or node.right:
                    not_all_none = True
            else:
                level.append(-1)
                queue.append(None)
                queue.append(None)
        result.append(level)
    return result


def print_tree(root, level=0, prefix='Root'):
    """美化打印 (左旋 90°)"""
    if not root:
        return
    print('  ' * level + f'{prefix}: {root.val}')
    print_tree(root.left, level + 1, 'L')
    print_tree(root.right, level + 1, 'R')
```

## A.3 四大基本遍历

### 模式 1: 前序遍历 (Pre-order: 根-左-右)

```python
def preorder_recur(root):
    """递归版"""
    res = []
    def dfs(node):
        if not node: return
        res.append(node.val)
        dfs(node.left)
        dfs(node.right)
    dfs(root)
    return res

def preorder_iter(root):
    """迭代版 (栈)"""
    res, stack = [], [root]
    while stack:
        node = stack.pop()
        if node:
            res.append(node.val)
            stack.append(node.right)  # 注意: 先右后左
            stack.append(node.left)
    return res

def preorder_morris(root):
    """Morris 遍历 - O(1) 空间 (修改树)"""
    res, cur = [], root
    while cur:
        if not cur.left:
            res.append(cur.val); cur = cur.right
        else:
            pre = cur.left
            while pre.right and pre.right is not cur:
                pre = pre.right
            if not pre.right:
                pre.right = cur
                res.append(cur.val); cur = cur.left
            else:
                pre.right = None
                cur = cur.right
    return res
```

```python
def postorder_recur(root):
    res = []
    def dfs(node):
        if not node: return
        dfs(node.left); dfs(node.right); res.append(node.val)
    dfs(root)
    return res

def postorder_iter(root):
    """迭代版: 前序变体 (根-右-左) 然后反转"""
    res, stack = [], [root]
    while stack:
        node = stack.pop()
        if node:

            stack.append(node.left)   # 注意: 先左后右
```

### 模式 4: 层序遍历 (Level-order: BFS)

```python
from collections import deque
def level_order(root):
    """每层一列表"""
    res = []
    if not root: return res
    q = deque([root])
    while q:
        level = []
        for _ in range(len(q)):
            node = q.popleft()
            level.append(node.val)
            if node.left: q.append(node.left)
            if node.right: q.append(node.right)
        res.append(level)
    return res

def level_order_2(root):
    """之字形打印 (锯齿形)"""
    if not root: return []
    res, q, reverse = [], deque([root]), False
    while q:
        level = []
        for _ in range(len(q)):
            node = q.popleft()
            level.append(node.val)
            if node.left: q.append(node.left)
            if node.right: q.append(node.right)
        if reverse:
            level.reverse()
        res.append(level)
        reverse = not reverse
    return res
```

```python
def maxDepth_bfs(root):
```

```python
# 上下界递归 (推荐)
def isValidBST(root):
    def validate(node, lo, hi):
        if not node: return True
        if node.val <= lo or node.val >= hi: return False
        return validate(node.left, lo, node.val) and validate(node.right, node.val, hi)

# 中序 + prev (更直观)
def isValidBST_inorder(root):
    self.prev = None
    def dfs(node):
        if not node: return True
        if not dfs(node.left): return False
        if self.prev is not None and node.val <= self.prev: return False
        self.prev = node.val
        return dfs(node.right)
```

## B.3 LCA 二叉树 (236) - 三大解法

```python
# 解法 1: 后序递归 (推荐)
def lowestCommonAncestor(root, p, q):
    if not root or root is p or root is q:
        return root
    L = lowestCommonAncestor(root.left, p, q)
    R = lowestCommonAncestor(root.right, p, q)
    if L and R: return root
    return L or R

# 解法 2: 父链哈希
def LCA_parent(root, p, q):
    def dfs(node, parent):
        if not node: return
        parent[node] = dfs(node.left, parent)
        parent[node] = (parent[node] or node)  # 占位
    parents = {root: None}
    def f(node):
        if not node: return
        if node.left: parents[node.left] = node
        if node.right: parents[node.right] = node
        f(node.left); f(node.right)
    f(root)

    ancestors = set()
    while p:
        ancestors.add(p)
        p = parents.get(p)
    while q:
        if q in ancestors: return q
        q = parents.get(q)

# 解法 3: BST 简化版 (235)
def LCA_BST(root, p, q):
    while root:
        if p.val < root.val and q.val < root.val:
            root = root.left
        elif p.val > root.val and q.val > root.val:
            root = root.right
        else: return root
```

```python
class BSTIterator:
    """解法 1: 模拟中序 (推荐)"""
    def __init__(self, root):
        self.stack = []
        self._push_left(root)

    def _push_left(self, node):
        while node:
            self.stack.append(node)
            node = node.left

    def next(self) -> int:
        node = self.stack.pop()
        self._push_left(node.right)
        return node.val

            if not node: return

        dfs(root)
        self.idx = -1

    def next(self) -> int:
        self.idx += 1
        return self.inorder[self.idx]

    def hasNext(self) -> bool:
```

## D.4 平衡 BST (AVL / Red-Black)

### AVL 树

```python
class AVLNode:
    __slots__ = ('val', 'left', 'right', 'height')
    def __init__(self, val):
        self.val = val
        self.height = 1
        self.left = self.right = None

def height(node):
    return node.height if node else 0

def update_height(node):
    if node:
        node.height = 1 + max(height(node.left), height(node.right))

def balance(node):
    return height(node.left) - height(node.right) if node else 0

def rotate_right(y):
    x = y.left
    T = x.right
    x.right = y
    y.left = T
    update_height(y); update_height(x)
    return x

def rotate_left(x):
    y = x.right
    T = y.left
    y.left = x
    x.right = T
    update_height(x); update_height(y)
    return y

def rebalance(node):
    balance_factor = balance(node)
    # LL
    if balance_factor > 1 and balance(node.left) >= 0:
        return rotate_right(node)
    # LR
    if balance_factor > 1 and balance(node.left) < 0:
        node.left = rotate_left(node.left)
        return rotate_right(node)
    # RR
    if balance_factor < -1 and balance(node.right) <= 0:
        return rotate_left(node)
    # RL
    if balance_factor < -1 and balance(node.right) > 0:
        node.right = rotate_right(node.right)
        return rotate_left(node)
    return node

def insert(node, val):
    if not node: return AVLNode(val)
    if val < node.val: node.left = insert(node.left, val)
    elif val > node.val: node.right = insert(node.right, val)
    update_height(node)
    return rebalance(node)
```

## Q1: 什么是二叉树的深度和高度?

- **节点的深度**: 根到节点的边数 (根深度=0)
- **节点的高度**: 节点到最远叶子的边数 (叶高度=0)
- **树的高度**: 根节点的高度

## Q2: 完美 / 完全 / 平衡 BST 的区别?

| 类型 | 定义 |
|------|------|
| 完美二叉树 | 所有叶在同一层, 节点满 |
| 完全二叉树 | 叶只在最后两层, 最后层左对齐 |
| 平衡二叉树 | 左右子树高度差 ≤ 1 |

## Q3: 为什么 BST 的中序是有序的?

BST 性质左子 < 父 < 右子, 中序"左-根-右" 遍历意味着:
按从小到大访问, 故中序 = 升序。

## Q4: 二叉树和 B 树有什么区别?

- 二叉树: 度 ≤ 2
- B 树: 度 > 2, 多键, 平衡, 用于数据库

## Q5: AVL vs Red-Black?

| 维度 | AVL | RB |
|------|-----|------|
| 平衡性 | 严格 | 弱平衡 |
| 查找 | O(log n) | O(log n) |
| 增删 | 多次旋转 | ≤ 3 次旋转 |
| 场景 | 查多 | 增删多 (STL/TreeMap) |

## Q6: LCA 三种解法的选择?

- 通用树 (236): 后序递归
- BST (235): 走子节点
- 父链已知: 哈希 + 上溯

## Q7: 如何用前序 + 中序构造唯一二叉树?

前序第一个是 root, 在中序找 root 切两半:
- 前序[1:mid+1] + 中序[:mid] → 左
- 前序[mid+1:] + 中序[mid+1:] → 右
(mid = 中序 root 索引)

## Q8: 为什么不能用前序 + 后序唯一构造?

因为已知 pre[0] = post[-1] = root, 但无法确定左子树在前序与后序中的边界。
例: pre = [1,2], post = [2,1] 既可生成单链右, 也可生成单链左。

但给定父子关系 (前序第二个是左子树根) 后, 即可。

## Q9: 二叉树的中序遍历能用 O(1) 空间吗?

可以! Morris 遍历 - 用叶子的 right 指针回溯, 时间 O(n)。

## Q10: 如何判断两棵二叉树相等?

**同构** (值 + 结构):
```python
def is_same(p, q):
    if not p and not q: return True
    if not p or not q: return False
    return p.val == q.val and is_same(p.left, q.left) and is_same(p.right, q.right)
```

## Q11: 如何判断两棵树同构 (572)?

```python
def isSubtree(s, t):

def is_same(p, q):
```

## Q12: 二叉树最大路径和 (124) 的本质?

路径可以拐弯, 形如根→左→根→右。
从左右子树"贡献"和"自带"两个角度:
- 全路径 = 左贡献 + 根 + 右贡献 (更新全局)
- 向上传递 = 根 + max(左贡献, 右贡献)

## Q13: 打家劫舍 III (337) 为什么用元组?

相邻节点不能同时偷, 每个节点面对两种选择:
- 不偷当前: 子树的最大值
- 偷当前: 当前值 + 子树不偷的最大值

元组 `(not_steal, steal)` 完美表达这两种状态。

## Q17: 求树的最大宽度 (662)?

BFS, 用 leftmost/rightmost index 差 + 1:
```python
def widthOfBinaryTree(root):
    if not root: return 0
    res, q = 1, deque([(root, 0)])
    while q:
        left = q[0][1]
        for _ in range(len(q)):
            node, idx = q.popleft()
            if node.left: q.append((node.left, idx*2))
            if node.right: q.append((node.right, idx*2+1))
        res = max(res, q[-1][1] - left + 1) if q else res
    return res
```

## Q18: 如何把二叉搜索树转双向链表?

中序遍历, prev/next 调指针:
```python
def tree_to_dll(root):
    self.head = self.pre = None
    def dfs(node):
        if not node: return
        dfs(node.left)
        if self.pre:
            self.pre.right = node; node.left = self.pre
        else:
            self.head = node
        self.pre = node
        dfs(node.right)
    dfs(root)
    return self.head
```

## Q19: 为什么 leaf-similar (367) 用栈?

```python
def leafSimilar(root1, root2):
    def get_leaves(root):
        if not root: return []
        if not root.left and not root.right: return [root.val]
        return get_leaves(root.left) + get_leaves(root.right)
    return get_leaves(root1) == get_leaves(root2)
```

## Q29: 节点最大的二叉树如何构造?

每次选最大值作根, 递归左右。
时间: O(n²) 朴素, O(n) 用单调栈。

```python
def pathSum(root, target):
    from collections import defaultdict
    cnt = defaultdict(int)
    cnt[0] = 1
    self.res = 0
    def dfs(node, prefix):
        if not node: return
        prefix += node.val
        self.res += cnt[prefix - target]
        cnt[prefix] += 1
        dfs(node.left, prefix)
        dfs(node.right, prefix)
        cnt[prefix] -= 1
    dfs(root, 0)
    return self.res

def subarraySum(nums, k):
    from collections import defaultdict
    cnt = defaultdict(int)
    cnt[0] = 1
    res, prefix = 0, 0
    for v in nums:
        prefix += v
        res += cnt[prefix - k]
        cnt[prefix] += 1
    return res
```

```python
    res, stack, cur = [], [], root
    while cur or stack:
        while cur:
            stack.append(cur); cur = cur.left
        cur = stack.pop()
```

## M.3 层序 BFS

```python
def bfs(root):
    from collections import deque
    res, q = [], deque([root])
    while q:
        level = []
        for _ in range(len(q)):
            node = q.popleft()
            level.append(node.val)
            if node.left: q.append(node.left)
            if node.right: q.append(node.right)
        res.append(level)
    return res
```

```python
def deleteBST(root, key):
    if not root: return None
    if key < root.val:
        root.left = deleteBST(root.left, key)
    elif key > root.val:
        root.right = deleteBST(root.right, key)
    else:
        if not root.left: return root.right
        if not root.right: return root.left
        cur = root.right
        while cur.left: cur = cur.left
        root.val = cur.val
        root.right = deleteBST(root.right, cur.val)
    return root
```

```python
    def dfs(node):
        if not node: return 0
        L = max(0, dfs(node.left))
        R = max(0, dfs(node.right))
        self.res = max(self.res, node.val + L + R)
```

## O.1 线段树 (Segment Tree)

```python
class SegTree:
    """区间和 / 区间最值"""
    def __init__(self, data):
        self.n = len(data)
        self.size = 1
        while self.size < self.n:
            self.size *= 2
        self.tree = [0] * (2 * self.size)
        for i in range(self.n):
            self.tree[self.size + i] = data[i]
        for i in range(self.size - 1, 0, -1):
            self.tree[i] = self.tree[2*i] + self.tree[2*i+1]

    def update(self, idx, val):
        i = self.size + idx
        self.tree[i] = val
        while i > 1:
            i //= 2
            self.tree[i] = self.tree[2*i] + self.tree[2*i+1]

    def query(self, l, r):
        """[l, r] 区间和"""
        res = 0
        l += self.size
        r += self.size + 1
        while l < r:
            if l % 2 == 1:
                res += self.tree[l]; l += 1
            if r % 2 == 1:
                r -= 1; res += self.tree[r]
            l //= 2; r //= 2
        return res
```

## O.2 字典树 (Trie)

```python
class Trie:
    def __init__(self):
        self.children = [None] * 26
        self.is_end = False

    def insert(self, word):
        node = self
        for c in word:
            idx = ord(c) - ord('a')
            if not node.children[idx]:
                node.children[idx] = Trie()
            node = node.children[idx]
        node.is_end = True

    def search(self, word):
        node = self
        for c in word:
            idx = ord(c) - ord('a')
            if not node.children[idx]: return False
            node = node.children[idx]
        return node.is_end

    def starts_with(self, prefix):
        node = self
        for c in prefix:
            idx = ord(c) - ord('a')
            if not node.children[idx]: return False
            node = node.children[idx]
        return True
```

## O.3 并查集 (Union-Find)

```python
class DSU:
    def __init__(self, n):
        self.par = list(range(n))
        self.rank = [0] * n

    def find(self, x):
        if self.par[x] != x:
            self.par[x] = self.find(self.par[x])
        return self.par[x]

    def union(self, a, b):
        ra, rb = self.find(a), self.find(b)
        if ra == rb: return False
        if self.rank[ra] < self.rank[rb]: ra, rb = rb, ra
        self.par[rb] = ra
        if self.rank[ra] == self.rank[rb]:
            self.rank[ra] += 1
        return True
```

**3. 数学形式化(简洁版)**

虽然我们强调思维框架视角,但数学形式化提供了共同语言。机器学习的核心问题可以表达为:给定输入样本,学习一个映射函数,使得函数输出与真实目标的差距最小化。这个抽象捕捉了分类、回归、生成等大多数任务的本质。

进一步,优化目标通常加入正则化项来控制模型复杂度,平衡经验风险与模型容量。统计学习理论给出了泛化误差的上界,告诉我们模型越复杂并不一定越好——**奥卡姆剃刀**在数学上有严格的形式化。

偏差-方差分解是另一个核心框架:误差可以分解为模型偏差(欠拟合)、方差(过拟合)和不可约噪声三部分。理解这个分解有助于诊断模型表现、选择合适复杂度。

**3. 数学形式化(简洁版)**

虽然我们强调边界条件视角,但数学形式化提供了共同语言。机器学习的核心问题可以表达为:给定输入样本,学习一个映射函数,使得函数输出与真实目标的差距最小化。这个抽象捕捉了分类、回归、生成等大多数任务的本质。

进一步,优化目标通常加入正则化项来控制模型复杂度,平衡经验风险与模型容量。统计学习理论给出了泛化误差的上界,告诉我们模型越复杂并不一定越好——**奥卡姆剃刀**在数学上有严格的形式化。

偏差-方差分解是另一个核心框架:误差可以分解为模型偏差(欠拟合)、方差(过拟合)和不可约噪声三部分。理解这个分解有助于诊断模型表现、选择合适复杂度。

在数值稳定性方面,有几个常见陷阱: 浮点数精度有限、Log-Sum-Exp 计算要减去最大值、Softmax 在大数值下会溢出。这些工程细节常常决定一个模型能否在生产环境稳定运行。

**4. 信息论视角**

信息论为"二叉树 (25 题)" 提供了统一的视角:**学习的本质是从数据中提取与目标相关的信息**。

互信息衡量知道一个变量后,另一个变量不确定性减少的量。在机器学习中,它被广泛用于特征选择、表示学习、多模态对齐等场景。

KL 散度衡量两个概率分布的距离,在交叉熵损失、变分推断、策略梯度等场景无处不在。需要注意它是**不对称**的——不同方向有不同的解释。

信息瓶颈原则指出,我们应该寻找既紧凑又预测性强的表示。这一思想影响了大量表示学习方法的设计,包括对比学习、自监督学习、多模态预训练等。

**7. 实践建议**

结合 "二叉树 (25 题)" 的特点,实践中有以下建议:

1. **数据优先**: 数据质量决定上限,模型只是逼近这个上限。把 50% 的时间投入到数据理解和清洗上,这条经验被无数实践证明。数据问题往往比模型问题更关键,但也更容易被忽视。

2. **基线先行**: 先用最简单的方法(如逻辑回归、规则)做基线,确认增益是来自方法还是数据。没有强基线的"创新"往往不靠谱。很多看似 fancy 的方法,在强基线面前提升有限。

3. **迭代式改进**: 不要追求一步到位,而是 baseline → 简单改进 → 高级方法 → 调优。每一步都用验证集确认有提升。每一步的提升幅度通常递减,要预期边际收益下降。

4. **失败案例分析**: 重点看模型做错的样本,往往能发现系统性问题。错误样本里藏着模型没学会的知识。把这些样本按错误类型分类,可以发现模型的薄弱点。

5. **A/B 测试**: 在生产环境做对照实验,用业务指标而非离线指标评估。离线指标只是手段,业务指标才是目的。指标可以被优化,但业务目标很难直接优化。

6. **监控与回滚**: 上线后持续监控关键指标,异常时快速回滚。任何线上系统都必须考虑"最坏情况",并预先设计好应对方案。

**8. 常见误解**

对 "二叉树 (25 题)" 常见的误解包括:

1. **"模型越复杂越好"**: 实际上,简单模型 + 好数据往往优于复杂模型 + 烂数据。这条法则在实践中反复被验证。

2. **"训练 loss 越低越好"**: 训练 loss 低不代表泛化好,可能过拟合。要看验证集表现,做误差分析。

3. **"指标好就等于业务好"**: 离线指标与在线业务有 gap,必须做 A/B 测试。指标可以被优化,但业务目标很难直接优化。

4. **"新方法一定比旧方法好"**: 新方法通常针对特定场景,在你的场景未必更好。要谨慎评估,先复现,再改进。

5. **"开源即用"**: 开源代码与你的数据/工程栈往往有 gap,需要适配。盲目复用经常踩坑,要理解代码再做改动。
